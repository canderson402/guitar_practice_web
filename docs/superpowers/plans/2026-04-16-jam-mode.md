import {
  getChromaticPosition,
  getScaleChords,
  getScaleNotes,
  chordTypes,
  scales,
} from './musicData';

// ---- Types ----------------------------------------------------------------

export type JamAlgorithm =
  | 'fifths' | 'fourths' | 'skip1' | 'skip2'
  | 'diatonic-fifths' | 'diatonic-fourths' | 'diatonic-thirds'
  | 'ii-v' | 'random';

export interface JamChord {
  note: string;       // 'C', 'Db', etc.
  type: string;       // key into chordTypes: 'major', 'minor', 'diminished'
  symbol: string;     // '', 'm', '°'
  roman: string;      // 'I', 'iv', 'V'
  midi: number[];     // e.g. [48, 52, 55]
}

// ---- Constants ------------------------------------------------------------

/** Circle of fifths — all 12 pitch classes in ascending-5ths order. */
const CIRCLE_OF_FIFTHS = [
  'C', 'G', 'D', 'A', 'E', 'B', 'Gb', 'Db', 'Ab', 'Eb', 'Bb', 'F',
];

/** Same circle reversed = cycle of fourths. */
const CIRCLE_OF_FOURTHS = [...CIRCLE_OF_FIFTHS].reverse();
// F, Bb, Eb, Ab, Db, Gb, B, E, A, D, G, C — then we rotate to start from C:
// Actually reverse of [C,G,D,A,E,B,Gb,Db,Ab,Eb,Bb,F] = [F,Bb,Eb,Ab,Db,Gb,B,E,A,D,G,C]
// We want fourths ascending from C: C,F,Bb,Eb,Ab,Db,Gb,B,E,A,D,G
const FOURTHS_FROM_C = (() => {
  const rev = [...CIRCLE_OF_FIFTHS].reverse(); // F,Bb,Eb,...,C
  const cIdx = rev.indexOf('C');
  return [...rev.slice(cIdx), ...rev.slice(0, cIdx)];
})();
```

// ---- MIDI derivation ------------------------------------------------------

/** Map a pitch class to a MIDI root in the comping range (C3–B3). */
const rootToMidi = (note: string): number => {
  const pc = getChromaticPosition(note);
  // Base octave 3 → C3 = MIDI 48. All roots land in [48..59].
  return 48 + pc;
};

/** Build the MIDI array for a chord given its root note and chord type key. */
export const chordToMidi = (
  note: string,
  type: string,
): number[] => {
  const root = rootToMidi(note);
  const intervals = (chordTypes as any)[type]?.intervals ?? [0, 4, 7]; // fallback major
  return intervals.map((i: number) => root + i);
};
```

// ---- Chord quality for non-diatonic notes ---------------------------------

/**
 * Determine the chord quality for a note in the context of a key/scale.
 * Uses parallel-mode borrowing: checks the selected scale, then parallel
 * minor, then harmonic minor, then falls back to major.
 *
 * Returns { type, symbol } from chordTypes.
 */
export const getChordQuality = (
  note: string,
  rootNote: string,
  scaleType: keyof typeof scales,
): { type: string; symbol: string } => {
  const notePos = getChromaticPosition(note);

  // Fallback chain of scales to check
  const fallbackScales: (keyof typeof scales)[] = [
    scaleType,
    'Aeolian (Natural Minor)',
    'Harmonic Minor',
  ];

  for (const s of fallbackScales) {
    const chords = getScaleChords(rootNote, s);
    const match = chords.find(c => getChromaticPosition(c.note) === notePos);
    if (match) {
      return { type: match.type, symbol: match.symbol };
    }
  }

  // Fallback: major
  return { type: 'major', symbol: '' };
};
```

// ---- Roman numeral labelling ----------------------------------------------

const ROMAN = ['I', 'II', 'III', 'IV', 'V', 'VI', 'VII'];

/**
 * Build a roman numeral label for a chord relative to the current key.
 * Diatonic chords get their proper numeral; chromatic ones get a flat prefix.
 */
const romanLabel = (
  note: string,
  type: string,
  rootNote: string,
  scaleType: keyof typeof scales,
): string => {
  const scaleNotes = getScaleNotes(rootNote, scaleType);
  const notePos = getChromaticPosition(note);

  // Check if it's a diatonic scale degree
  const degreeIdx = scaleNotes.findIndex(n => getChromaticPosition(n) === notePos);

  let roman: string;
  if (degreeIdx !== -1) {
    roman = ROMAN[degreeIdx] ?? `${degreeIdx + 1}`;
  } else {
    // Chromatic: find the nearest diatonic degree above and label as flat
    // e.g. Db in C major → bII
    const chromaticScale = ['C','C#','D','D#','E','F','F#','G','G#','A','A#','B'];
    const rootPos = getChromaticPosition(rootNote);
    const semitones = (notePos - rootPos + 12) % 12;
    // Map semitone distance to the closest diatonic degree name
    // 1→bII, 3→bIII, 6→bV/#IV, 8→bVI, 10→bVII
    const semitoneToDegree: Record<number, string> = {
      1: 'bII', 3: 'bIII', 6: '#IV', 8: 'bVI', 10: 'bVII',
    };
    roman = semitoneToDegree[semitones] ?? `b${ROMAN[Math.round(semitones * 7 / 12)] ?? '?'}`;
  }

  // Lowercase for minor/diminished
  if (type === 'minor' || type === 'diminished') {
    roman = roman.toLowerCase();
  }
  if (type === 'diminished') {
    roman += '°';
  }

  return roman;
};

// ---- Chromatic walk algorithms --------------------------------------------

/** Step through the 12-note circle with a given skip size.
 *  skip=0 → every note, skip=1 → every other, skip=2 → every third, etc. */
const chromaticWalk = (
  circle: string[],
  currentNote: string,
  skip: number,
): string => {
  const pos = getChromaticPosition(currentNote);
  const idx = circle.findIndex(n => getChromaticPosition(n) === pos);
  const step = skip + 1;
  const nextIdx = (idx + step) % circle.length;
  return circle[nextIdx >= 0 ? nextIdx : nextIdx + circle.length];
};

const nextChordChromatic = (
  algorithm: 'fifths' | 'fourths' | 'skip1' | 'skip2',
  currentNote: string,
): string => {
  switch (algorithm) {
    case 'fifths':  return chromaticWalk(CIRCLE_OF_FIFTHS, currentNote, 0);
    case 'fourths': return chromaticWalk(FOURTHS_FROM_C, currentNote, 0);
    case 'skip1':   return chromaticWalk(CIRCLE_OF_FIFTHS, currentNote, 1);
    case 'skip2':   return chromaticWalk(CIRCLE_OF_FIFTHS, currentNote, 2);
  }
};
```

```ts
// ---- Diatonic walk algorithms ---------------------------------------------

const nextChordDiatonic = (
  algorithm: 'diatonic-fifths' | 'diatonic-fourths' | 'diatonic-thirds' | 'ii-v' | 'random',
  currentNote: string,
  rootNote: string,
  scaleType: keyof typeof scales,
  /** For ii-V, tracks position within the current ii-V-I group (0=ii, 1=V, 2=I). */
  iiVState?: { position: number; targetDegree: number },
): { note: string; iiVState?: { position: number; targetDegree: number } } => {
  const scaleChords = getScaleChords(rootNote, scaleType);
  if (scaleChords.length === 0) return { note: currentNote };

  const n = scaleChords.length; // typically 7
  const currentPos = getChromaticPosition(currentNote);
  const currentIdx = scaleChords.findIndex(c => getChromaticPosition(c.note) === currentPos);
  // If current note isn't diatonic, snap to nearest
  const safeIdx = currentIdx === -1 ? 0 : currentIdx;

  switch (algorithm) {
    case 'diatonic-fifths': {
      // +4 scale degrees = diatonic 5th
      const nextIdx = (safeIdx + 4) % n;
      return { note: scaleChords[nextIdx].note };
    }
    case 'diatonic-fourths': {
      // +3 scale degrees = diatonic 4th
      const nextIdx = (safeIdx + 3) % n;
      return { note: scaleChords[nextIdx].note };
    }
    case 'diatonic-thirds': {
      // +2 scale degrees = diatonic 3rd
      const nextIdx = (safeIdx + 2) % n;
      return { note: scaleChords[nextIdx].note };
    }
    case 'ii-v': {
      // Walk ii-V-I groups targeting each diatonic chord in order.
      // Group cycle: for target degree D, play chord at (D-2), (D-1), D.
      // That's: ii of D → V of D → D itself.
      if (!iiVState) {
        // Start fresh: target degree 0 (the I chord), position 0 (the ii)
        const iiIdx = (0 + n - 2) % n; // ii relative to I
        return {
          note: scaleChords[iiIdx].note,
          iiVState: { position: 1, targetDegree: 0 },
        };
      }
      const { position, targetDegree } = iiVState;
      if (position === 1) {
        // Play V of target
        const vIdx = (targetDegree + n - 1) % n;
        return {
          note: scaleChords[vIdx].note,
          iiVState: { position: 2, targetDegree },
        };
      }
      // position === 2: play the target itself, then advance target
      const nextTarget = (targetDegree + 1) % n;
      const note = scaleChords[targetDegree].note;
      // Next call starts a new group: ii of nextTarget
      return {
        note,
        iiVState: { position: 0, targetDegree: nextTarget },
      };
    }
    case 'random': {
      let nextIdx: number;
      do {
        nextIdx = Math.floor(Math.random() * n);
      } while (nextIdx === safeIdx && n > 1);
      return { note: scaleChords[nextIdx].note };
    }
  }
};
```

```ts
// ---- Public API -----------------------------------------------------------

const CHROMATIC_ALGORITHMS = new Set<JamAlgorithm>([
  'fifths', 'fourths', 'skip1', 'skip2',
]);

export interface WalkState {
  iiVState?: { position: number; targetDegree: number };
}

/**
 * Generate the next chord in the walk.
 *
 * Pure function — takes everything it needs as arguments, returns the next
 * JamChord plus any updated walk state (currently only ii-V needs state).
 */
export const generateNextChord = (
  algorithm: JamAlgorithm,
  currentNote: string,
  rootNote: string,
  scaleType: keyof typeof scales,
  walkState: WalkState = {},
): { chord: JamChord; walkState: WalkState } => {
  let nextNote: string;
  let newWalkState: WalkState = {};

  if (CHROMATIC_ALGORITHMS.has(algorithm)) {
    nextNote = nextChordChromatic(
      algorithm as 'fifths' | 'fourths' | 'skip1' | 'skip2',
      currentNote,
    );
  } else {
    const result = nextChordDiatonic(
      algorithm as 'diatonic-fifths' | 'diatonic-fourths' | 'diatonic-thirds' | 'ii-v' | 'random',
      currentNote,
      rootNote,
      scaleType,
      walkState.iiVState,
    );
    nextNote = result.note;
    if (result.iiVState) {
      newWalkState.iiVState = result.iiVState;
    }
  }

  const chord = buildJamChord(nextNote, rootNote, scaleType);
  return { chord, walkState: newWalkState };
};
```

// ---- Preset queue builder -------------------------------------------------

/**
 * Resolve a progression preset (array of roman numeral strings like ['I','IV','V7'])
 * into concrete JamChords in the given key.
 *
 * Roman numerals are matched against getScaleChords() output. Suffixed qualities
 * (e.g. 'V7', 'ii7', 'Imaj7') are parsed: the base roman maps to a scale degree,
 * then the suffix overrides the chord type.
 */
export const buildPresetQueue = (
  romanNumerals: string[],
  rootNote: string,
  scaleType: keyof typeof scales,
): JamChord[] => {
  const scaleChords = getScaleChords(rootNote, scaleType);
  if (scaleChords.length === 0) return [];

  const romanToIdx: Record<string, number> = {
    'I': 0, 'II': 1, 'III': 2, 'IV': 3, 'V': 4, 'VI': 5, 'VII': 6,
  };

  // Suffix → chordTypes key
  const suffixToType: Record<string, { type: string; symbol: string }> = {
    '7':    { type: 'dominant7', symbol: '7' },
    'maj7': { type: 'major7', symbol: 'maj7' },
    'maj9': { type: 'major7', symbol: 'maj7' },   // treat as maj7 for MVP
    'm7':   { type: 'minor7', symbol: 'm7' },
    'm9':   { type: 'minor7', symbol: 'm7' },     // treat as m7 for MVP
    '°7':   { type: 'diminished7', symbol: '°7' },
    'ø7':   { type: 'half-diminished7', symbol: 'ø7' },
    '°':    { type: 'diminished', symbol: '°' },
    '6':    { type: 'major', symbol: '' },          // treat as major for MVP
    'i6':   { type: 'minor', symbol: 'm' },
    '9':    { type: 'dominant7', symbol: '7' },     // treat as dom7 for MVP
    '13':   { type: 'dominant7', symbol: '7' },     // treat as dom7 for MVP
  };

  return romanNumerals.map(raw => {
    // Handle special cases like 'bVII', 'bVI', '#IV', 'I/V' (slash chords → use root)
    let cleaned = raw.replace(/\/.*$/, ''); // strip slash-bass for MVP

    // Extract flat/sharp prefix
    let prefix = '';
    if (cleaned.startsWith('b') || cleaned.startsWith('#')) {
      prefix = cleaned[0];
      cleaned = cleaned.slice(1);
    }

    // Separate base roman from suffix: 'Imaj7' → base='I', suffix='maj7'
    // 'ii7' → base='ii', suffix='7'
    // 'V7' → base='V', suffix='7'
    const baseMatch = cleaned.match(/^([iIvV]+)(.*)/);
    if (!baseMatch) {
      // Fallback — can't parse, return root chord
      return buildJamChord(rootNote, rootNote, scaleType);
    }

    const baseRoman = baseMatch[1];
    const suffix = baseMatch[2]; // e.g. '7', 'maj7', '°7', ''

    const upperRoman = baseRoman.toUpperCase();
    const degreeIdx = romanToIdx[upperRoman];
    if (degreeIdx === undefined) {
      return buildJamChord(rootNote, rootNote, scaleType);
    }

    // Handle flat/sharp adjustment
    let chordNote: string;
    if (prefix === 'b' || prefix === '#') {
      const scaleNotes = getScaleNotes(rootNote, scaleType);
      const basePos = getChromaticPosition(scaleNotes[degreeIdx] ?? rootNote);
      const adjusted = prefix === 'b' ? (basePos - 1 + 12) % 12 : (basePos + 1) % 12;
      const chromatic = ['C', 'C#', 'D', 'D#', 'E', 'F', 'F#', 'G', 'G#', 'A', 'A#', 'B'];
      chordNote = chromatic[adjusted];
    } else if (degreeIdx < scaleChords.length) {
      chordNote = scaleChords[degreeIdx].note;
    } else {
      chordNote = rootNote;
    }

    // Determine chord type
    let type: string;
    let symbol: string;
    if (suffix && suffixToType[suffix]) {
      type = suffixToType[suffix].type;
      symbol = suffixToType[suffix].symbol;
    } else if (suffix === '' && degreeIdx < scaleChords.length && !prefix) {
      // No suffix, diatonic — use the scale-derived quality
      type = scaleChords[degreeIdx].type;
      symbol = scaleChords[degreeIdx].symbol;
    } else {
      // Has a prefix (bVII etc) or unknown suffix — resolve via quality chain
      const quality = getChordQuality(chordNote, rootNote, scaleType);
      type = quality.type;
      symbol = quality.symbol;
    }

    const roman = romanLabel(chordNote, type, rootNote, scaleType);
    const midi = chordToMidi(chordNote, type);
    return { note: chordNote, type, symbol, roman, midi };
  });
};
```

  })),

  advanceJamChord: () => set((state) => {
    const { jam, note } = state;
    const queue = [...jam.chordQueue];
    let nextIndex = jam.currentChordIndex + 1;
    let walkState = jam.walkState;

    if (jam.mode === 'preset') {
      // Loop: wrap around
      if (nextIndex >= queue.length) nextIndex = 0;
    } else {
      // Infinite: append new chord at tail to keep the queue fed
      if (queue.length - nextIndex < jam.queueLength) {
        const lastNote = queue.length > 0 ? queue[queue.length - 1].note : (note.selectedNote ?? 'C');
        const rootNote = note.selectedNote ?? 'C';
        const scaleType = (note.selectedScale ?? 'Major (Ionian)') as keyof typeof scales;
        const result = generateNextChord(jam.algorithm, lastNote, rootNote, scaleType, walkState);
        queue.push(result.chord);
        walkState = result.walkState;
      }
    }

    const current = queue[nextIndex];
    if (!current) return state;

    return {
      jam: { ...jam, currentChordIndex: nextIndex, chordQueue: queue, walkState },
      // Atomic update — avoids the selectedChord-clearing in setSelectedNote
      note: {
        ...note,
        selectedNote: current.note,
        selectedChord: {
          note: current.note,
          type: current.type,
          symbol: current.symbol,
          roman: current.roman,
        },
      },
    };
  }),

  rebuildJamQueue: () => set((state) => {
    const { jam, note } = state;
    const rootNote = note.selectedNote ?? 'C';
    const scaleType = (note.selectedScale ?? 'Major (Ionian)') as keyof typeof scales;

    let queue: JamChord[] = [];
    let walkState: WalkState = {};

    if (jam.mode === 'preset' && jam.selectedPreset) {
      const pool = { ...majorProgressions, ...minorProgressions } as Record<string, { chords: string[] }>;
      const preset = pool[jam.selectedPreset];
      if (preset) {
        queue = buildPresetQueue(preset.chords, rootNote, scaleType);
      }
    } else {
      // Infinite: seed the queue with queueLength chords starting from the root
      let currentNote = rootNote;
      const firstChord = buildJamChord(rootNote, rootNote, scaleType);
      queue.push(firstChord);
      for (let i = 1; i < jam.queueLength; i++) {
        const result = generateNextChord(jam.algorithm, currentNote, rootNote, scaleType, walkState);
        queue.push(result.chord);
        currentNote = result.chord.note;
        walkState = result.walkState;
      }
    }

    const current = queue[0];
    return {
      jam: { ...jam, chordQueue: queue, currentChordIndex: 0, walkState },
      note: current ? {
        ...state.note,
        selectedNote: current.note,
        selectedChord: {
          note: current.note,
          type: current.type,
          symbol: current.symbol,
          roman: current.roman,
        },
      } : state.note,
    };
  }),
```

import React, { useEffect, useRef, useCallback } from 'react';
import { useStore } from '../store/useStore';
import {
  majorProgressions,
  minorProgressions,
  scales,
} from '../data/musicData';
import { drumPatterns } from '../data/jamAlgorithms';
import type { JamAlgorithm, DrumPatternName } from '../data/jamAlgorithms';
import { createScheduler, playChord, playKick, playSnare, playHat } from '../audio';
import type { Scheduler } from '../audio';
import { Button, Select, Checkbox, ToggleButtonGroup } from '../ui';
import './JamCard.css';

const ALGORITHM_OPTIONS: { value: JamAlgorithm; label: string; group: string }[] = [
  { value: 'fifths', label: 'Fifths', group: 'Chromatic' },
  { value: 'fourths', label: 'Fourths', group: 'Chromatic' },
  { value: 'skip1', label: 'Skip 1 (whole tone)', group: 'Chromatic' },
  { value: 'skip2', label: 'Skip 2 (minor 3rds)', group: 'Chromatic' },
  { value: 'diatonic-fifths', label: 'Diatonic Fifths', group: 'Diatonic' },
  { value: 'diatonic-fourths', label: 'Diatonic Fourths', group: 'Diatonic' },
  { value: 'diatonic-thirds', label: 'Diatonic Thirds', group: 'Diatonic' },
  { value: 'ii-v', label: 'ii-V Pairs', group: 'Diatonic' },
  { value: 'random', label: 'Random', group: 'Diatonic' },
];

const DRUM_PATTERN_OPTIONS: { value: DrumPatternName; label: string }[] = [
  { value: 'rock', label: 'Rock' },
  { value: 'bossa', label: 'Bossa' },
  { value: 'hiphop', label: 'Hip-hop' },
];

export const JamCard: React.FC = () => {
  const {
    jam,
    note,
    setJamMode,
    setJamPlaying,
    setJamBpm,
    setJamPreset,
    setJamAlgorithm,
    setJamDrumsEnabled,
    setJamDrumPattern,
    setJamBeatsPerChord,
    setJamSyncMetronome,
    rebuildJamQueue,
    setMetronomePlaying,
    setBpm,
    setCurrentBeat,
  } = useStore();

  const schedulerRef = useRef<Scheduler | null>(null);

  // Build/rebuild queue when mode, preset, algorithm, key, or scale changes
  const prevDepsRef = useRef<string>('');
  useEffect(() => {
    const depsKey = `${jam.mode}|${jam.selectedPreset}|${jam.algorithm}|${note.selectedNote}|${note.selectedScale}`;
    if (depsKey !== prevDepsRef.current) {
      prevDepsRef.current = depsKey;
      rebuildJamQueue();
    }
  }, [jam.mode, jam.selectedPreset, jam.algorithm, note.selectedNote, note.selectedScale, rebuildJamQueue]);

  const startPlayback = useCallback(() => {
    const scheduler = createScheduler(jam.bpm, (beatIndex, audioTime) => {
      // Read fresh state directly from Zustand — NOT from a ref or closure.
      // The scheduler callback fires from setInterval, outside React lifecycle,
      // so stale closures are the #1 bug source here.
      const state = useStore.getState();
      const j = state.jam;
      const beatsPerBar = 4; // 4/4 time
      const beatInBar = beatIndex % beatsPerBar;
      const beatInChord = beatIndex % j.beatsPerChord;

      // Update metronome beat display
      if (j.syncMetronome) {
        state.setCurrentBeat(beatInBar);
      }

      // Chord handling on first beat of each chord group
      if (beatInChord === 0) {
        // Advance first (skip beat 0 — rebuildJamQueue already set chord 0)
        if (beatIndex > 0) {
          state.advanceJamChord();
        }
        // Read fresh state AFTER advance so we get the new chord
        const fresh = useStore.getState();
        const currentChord = fresh.jam.chordQueue[fresh.jam.currentChordIndex];
        if (currentChord) {
          const chordDuration = (60 / j.bpm) * j.beatsPerChord * 0.9;
          playChord(currentChord.midi, chordDuration, audioTime, { gain: 0.2, waveform: 'sawtooth', brightness: 0.3 });
        }
      }

      // Play drums
      if (j.drumsEnabled) {
        const pattern = drumPatterns[j.drumPattern];
        const beatDuration = 60 / j.bpm;
        for (const hit of pattern) {
          const hitBeat = Math.floor(hit.position);
          const hitOffset = hit.position - hitBeat;
          if (hitBeat === beatInBar) {
            const hitTime = audioTime + hitOffset * beatDuration;
            switch (hit.instrument) {
              case 'kick':  playKick(hitTime, hit.gain); break;
              case 'snare': playSnare(hitTime, hit.gain); break;
              case 'hat':   playHat(hitTime, true, hit.gain); break;
            }
          }
        }
      }
    });

    schedulerRef.current = scheduler;
    scheduler.start();
    setJamPlaying(true);

    if (jam.syncMetronome) {
      setMetronomePlaying(true);
      setBpm(jam.bpm);
    }
  }, [jam.bpm, jam.syncMetronome, setJamPlaying, setMetronomePlaying, setBpm]);

  const stopPlayback = useCallback(() => {
    schedulerRef.current?.stop();
    schedulerRef.current = null;
    setJamPlaying(false);
    beatCountRef.current = 0;

    if (jam.syncMetronome) {
      setMetronomePlaying(false);
      setCurrentBeat(0);
    }
  }, [jam.syncMetronome, setJamPlaying, setMetronomePlaying, setCurrentBeat]);

  // Update scheduler BPM when it changes mid-play
  useEffect(() => {
    if (schedulerRef.current?.isRunning()) {
      schedulerRef.current.setBpm(jam.bpm);
    }
  }, [jam.bpm]);

  // Cleanup on unmount
  useEffect(() => {
    return () => {
      schedulerRef.current?.stop();
    };
  }, []);

  const handleBpmChange = (delta: number) => {
    setJamBpm(jam.bpm + delta);
  };

  // Build preset options from progressions
  const progressionPool = jam.mode === 'preset'
    ? (note.selectedScale && ['Aeolian (Natural Minor)', 'Dorian', 'Phrygian', 'Locrian', 'Harmonic Minor', 'Minor Pentatonic'].includes(note.selectedScale)
        ? minorProgressions
        : majorProgressions)
    : {};

  const presetOptions = Object.entries(progressionPool).map(([key, val]) => ({
    value: key,
    label: key,
  }));

  // Current chord display
  const currentChord = jam.chordQueue[jam.currentChordIndex];
  const upcomingChords = jam.chordQueue.slice(jam.currentChordIndex + 1, jam.currentChordIndex + 6);

  return (
    <div className="jam-card">
      {/* Mode Toggle */}
      <div className="jam-mode-toggle">
        <ToggleButtonGroup label="Mode" layout="segmented">
          <Button
            variant="ghost"
            size="sm"
            active={jam.mode === 'preset'}
            onClick={() => setJamMode('preset')}
          >
            Preset
          </Button>
          <Button
            variant="ghost"
            size="sm"
            active={jam.mode === 'infinite'}
            onClick={() => setJamMode('infinite')}
          >
            Infinite
          </Button>
        </ToggleButtonGroup>
      </div>

      {/* Mode-specific picker */}
      {jam.mode === 'preset' ? (
        <div className="jam-picker">
          <label className="ds-label-inline">Progression:</label>
          <Select
            size="sm"
            value={jam.selectedPreset ?? ''}
            onChange={(e) => setJamPreset(e.target.value || null)}
            options={[
              { value: '', label: 'Choose...' },
              ...presetOptions,
            ]}
          />
        </div>
      ) : (
        <div className="jam-picker">
          <label className="ds-label-inline">Algorithm:</label>
          <Select
            size="sm"
            value={jam.algorithm}
            onChange={(e) => setJamAlgorithm(e.target.value as JamAlgorithm)}
            options={ALGORITHM_OPTIONS.map(o => ({ value: o.value, label: `${o.group}: ${o.label}` }))}
          />
        </div>
      )}

      {/* Chord Display */}
      <div className="jam-chord-display">
        <div className="jam-current-chord">
          {currentChord ? (
            <>
              <div className="jam-chord-note">{currentChord.note}{currentChord.symbol}</div>
              <div className="jam-chord-roman">{currentChord.roman}</div>
            </>
          ) : (
            <div className="jam-chord-note">—</div>
          )}
        </div>
        <div className="jam-upcoming">
          {upcomingChords.map((chord, i) => (
            <div key={`${chord.note}-${chord.type}-${i}`} className="jam-upcoming-chord" style={{ opacity: 1 - i * 0.15 }}>
              <div className="jam-upcoming-note">{chord.note}{chord.symbol}</div>
              <div className="jam-upcoming-roman">{chord.roman}</div>
            </div>
          ))}
        </div>
      </div>

      {/* Controls Row: Drums */}
      <div className="jam-controls-row">
        <Checkbox
          checked={jam.drumsEnabled}
          onCheckedChange={setJamDrumsEnabled}
          label="Drums"
        />
        <Select
          size="sm"
          value={jam.drumPattern}
          onChange={(e) => setJamDrumPattern(e.target.value as DrumPatternName)}
          options={DRUM_PATTERN_OPTIONS}
          disabled={!jam.drumsEnabled}
        />
      </div>

      {/* Controls Row: BPM */}
      <div className="jam-bpm-row">
        <Button variant="outline" size="sm" onClick={() => handleBpmChange(-5)}>−5</Button>
        <Button variant="outline" size="sm" onClick={() => handleBpmChange(-1)}>−1</Button>
        <div className="jam-bpm-display">
          <span className="jam-bpm-value">{jam.bpm}</span>
          <span className="jam-bpm-label">BPM</span>
        </div>
        <Button variant="outline" size="sm" onClick={() => handleBpmChange(1)}>+1</Button>
        <Button variant="outline" size="sm" onClick={() => handleBpmChange(5)}>+5</Button>
      </div>

      {/* Controls Row: Beats per chord + Sync */}
      <div className="jam-controls-row">
        <div className="jam-control-compact">
          <label className="ds-label-inline">Beats/chord:</label>
          <Select
            size="sm"
            value={jam.beatsPerChord}
            onChange={(e) => setJamBeatsPerChord(Number(e.target.value))}
            options={[1,2,3,4,5,6,7,8].map(n => ({ value: String(n), label: String(n) }))}
          />
        </div>
        <Checkbox
          checked={jam.syncMetronome}
          onCheckedChange={setJamSyncMetronome}
          label="Sync Metronome"
        />
      </div>

      {/* Play / Stop */}
      <Button
        variant={jam.isPlaying ? 'danger' : 'primary'}
        onClick={() => jam.isPlaying ? stopPlayback() : startPlayback()}
        disabled={jam.mode === 'preset' && !jam.selectedPreset}
      >
        {jam.isPlaying ? 'Stop' : 'Play'}
      </Button>
    </div>
  );
};
